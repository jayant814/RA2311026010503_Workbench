cmake_minimum_required(VERSION 3.16)
project(ThermoBridge VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(core src/core/thermal.cpp src/core/import.cpp src/core/experiments.cpp src/core/hardware.cpp)
target_include_directories(core PUBLIC src)
target_link_libraries(core PUBLIC Threads::Threads)
if(MSVC)
 target_compile_options(core PRIVATE /W4 /permissive-)
else()
 target_compile_options(core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(workbench_cli src/cli.cpp)
target_link_libraries(workbench_cli PRIVATE core)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE core)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
set_tests_properties(core_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
if(WIN32)
 add_executable(Launch WIN32 src/ui/winmain.cpp)
 target_link_libraries(Launch PRIVATE core comdlg32 gdi32 user32 shell32 advapi32)
 target_compile_definitions(Launch PRIVATE UNICODE _UNICODE NOMINMAX)
 if(MINGW)
  target_link_options(Launch PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(workbench_cli PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
endif()
if(WIN32)
 target_link_libraries(core PUBLIC advapi32)
endif()
